<script>
    // 故事生成邏輯
    function generateStory() {
        // 1. 獲取使用者輸入的數值 (若未輸入則使用預設值)
        const person = document.getElementById('person').value.trim() || '神秘主角';
        const time = document.getElementById('time').value.trim() || '很久很久以前';
        const place = document.getElementById('place').value.trim() || '未知的角落';
        const action = document.getElementById('action').value.trim() || '進行一場冒險';
        const item = document.getElementById('item').value.trim() || '神秘的物品';

        // 2. 組成故事模板 (可在此自訂故事內文與邏輯)
        const story = `在${time}，${person}獨自來到${place}。在那裡，${person}正在嘗試${action}。就在這個關鍵時刻，${person}意外發現了手中的${item}發出了微弱的光芒，這將永遠改變接下來發生的一切...`;

        // 3. 將故事渲染到網頁上
        document.getElementById('storyOutput').innerText = story;
    }
</script>
